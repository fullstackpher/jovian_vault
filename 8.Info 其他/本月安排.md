## 今日安排

```tasks
not done
due on today
```


```dataviewjs
const today = dv.date("today");
const weekEnd = dv.date("today").plus({days: 6});

// 查询任务
const todayTasks = dv.pages()
  .where(p => p.file.tasks)
  .file.tasks
  .where(t => !t.completed && t.due && t.due.toFormat("yyyy-MM-dd") === today.toFormat("yyyy-MM-dd"));

const weekTasks = dv.pages()
  .where(p => p.file.tasks)
  .file.tasks
  .where(t => !t.completed && t.due && t.due >= today && t.due <= weekEnd);

// 创建更美观的面板
dv.span(`
<div class="task-panel">
  <div class="task-section">
    <h3>今天要完成的任务</h3>
    <div class="task-count">${todayTasks.length} tasks</div>
    <div class="task-list">
      ${todayTasks.length > 0 
        ? todayTasks.values.map(t => `<div class="task-item">• ${t.text}</div>`).join("") 
        : '<div class="no-task">暂无任务</div>'}
    </div>
  </div>
  
  <div style="margin: 20px 0;"></div>
  
  <div class="task-section">
    <h3>一周内要完成的任务</h3>
    <div class="task-count">${weekTasks.length} tasks</div>
    <div class="task-list">
      ${weekTasks.length > 0 
        ? weekTasks.values.map(t => `<div class="task-item">• ${t.text} <span class="due-date">(${t.due.toFormat("MM-dd")})</span></div>`).join("") 
        : '<div class="no-task">暂无任务</div>'}
    </div>
  </div>
</div>

<style>
```

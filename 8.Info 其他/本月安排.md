## 今日安排

```tasks
not done
due on today
```


```dataviewjs
const today = dv.date("today");
const weekEnd = dv.date("today").plus({days: 6});

// 查询今日任务
const todayTasks = dv.pages()
  .where(p => p.file.tasks)
  .file.tasks
  .where(t => !t.completed && t.due && t.due.toFormat("yyyy-MM-dd") === today.toFormat("yyyy-MM-dd"));

// 查询一周内任务（不包括今天）
const weekTasks = dv.pages()
  .where(p => p.file.tasks)
  .file.tasks
  .where(t => !t.completed && t.due && 
         t.due > today && t.due <= weekEnd);

// 创建Markdown格式的输出
let output = "";

// 今日任务
output += "## 今天要完成的任务\n";
output += `${todayTasks.length} tasks\n\n`;
if (todayTasks.length > 0) {
  todayTasks.values.forEach(t => {
    output += `- ${t.text}\n`;
  });
} else {
  output += "> 暂无任务\n";
}

output += "\n---\n\n";

// 一周内任务
output += "## 一周内要完成的任务\n";
output += `${weekTasks.length} tasks\n\n`;
if (weekTasks.length > 0) {
  weekTasks.values.forEach(t => {
    const dueDate = t.due.toFormat("MM-dd");
    output += `- ${t.text} (${dueDate})\n`;
  });
} else {
  output += "> 暂无任务\n";
}

dv.paragraph(output);
```

## 今日安排

```tasks
not done
due on today
```


```dataviewjs
const today = dv.date("today");
const weekEnd = dv.date("today").plus({days: 6});

// 查询今日任务
const todayTasks = dv.pages()
  .where(p => p.file.tasks)
  .file.tasks
  .where(t => !t.completed && t.due && t.due.toFormat("yyyy-MM-dd") === today.toFormat("yyyy-MM-dd"));

// 查询一周内任务（包括今天）
const weekTasks = dv.pages()
  .where(p => p.file.tasks)
  .file.tasks
  .where(t => !t.completed && t.due && 
         t.due >= today && t.due <= weekEnd);

// 创建面板
const createTaskPanel = (title, tasks, period) => {
  const count = tasks.length;
  const taskList = tasks.length > 0 
    ? tasks.values.map(t => `- ${t.text}`).join("\n")
    : "";

  return `
## ${title}
${count} tasks
${taskList}
`;
};

// 输出面板
dv.span(`
${createTaskPanel("今天要完成的任务", todayTasks, "today")}

${createTaskPanel("一周内要完成的任务", weekTasks, "week")}
`);
```

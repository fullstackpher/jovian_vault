```dataviewjs
// 使用网络一言API获取每日一言
(async () => {
  try {
    // hitokoto API: c=i 文学, c=n 诗词, c=k 台词
    const response = await fetch('https://v1.hitokoto.cn/?c=i&c=n&c=k');
    const data = await response.json();
    const content = data.hitokoto;
    const from = data.from || data.from_who || '未知';
    dv.paragraph(`> [!quote] ✨ 一言\n${content}\n\n—— ${from}`);
  } catch (e) {
    // 网络失败时使用备用一言
    const 备用一言 = [
      "千里之行，始于足下 —— 老子",
      "路漫漫其修远兮，吾将上下而求索 —— 屈原",
      "学而不思则罔，思而不学则殆 —— 孔子",
      "业精于勤，荒于嬉；行成于思，毁于随 —— 韩愈",
      "纸上得来终觉浅，绝知此事要躬行 —— 陆游"
    ];
    const fileDate = dv.current().file.cday;
    const seed = fileDate.year * 10000 + fileDate.month * 100 + fileDate.day;
    dv.paragraph(`> [!quote] ✨ 一言\n${备用一言[seed % 备用一言.length]}`);
  }
})();
```

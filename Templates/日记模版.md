// 使用备用API列表，提高稳定性
const apis = [
  'https://v1.hitokoto.cn/?c=d&c=h&c=i&c=j',
  'https://hitokoto.cn/api?type=d',
  'https://api.shadiao.pro/chp'
];
let 一言 = "加载中...";
let 来源 = "佚名的";
let 作者 = "";

async function fetchWithTimeout(url, timeout = 3000) {
  const controller = new AbortController();
  const id = setTimeout(() => controller.abort(), timeout);
  try {
    const response = await fetch(url, { signal: controller.signal });
    clearTimeout(id);
    return response;
  } catch (e) {
    clearTimeout(id);
    throw e;
  }
}

try {
  let data = null;
  for (const api of apis) {
    try {
      const response = await fetchWithTimeout(api);
      data = await response.json();
      break;
    } catch (e) {
      continue;
    }
  }

  if (data) {
    一言 = data.hitokoto || data.text || "今日一言获取失败";
    来源 = data.from || data.from_who || "未知来源";
    作者 = data.from_who || '佚名的';
  } else {
    一言 = "今日一言服务暂时不可用";
  }
} catch (error) {
  console.error("获取一言失败:", error);
  一言 = "今日一言加载失败";
}

dv.paragraph(`> [!quote] ✨ 一言\n${一言} —— ​《${来源}》 · ${作者}`);
